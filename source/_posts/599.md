# 清除登陆系统成功的记录

```c
echo > /var/log/wtmp #此文件默认打开时乱码，可查到ip等信息
last #此时即查不到用户登录信息
```

# 清除登陆系统失败的记录

```c
echo > /var/log/btmp    #此文件默认打开时乱码，可查到登陆失败信息
lastb                     #查不到登陆失败信息
```

# 清除历史执行命令

```c
history -c # 清空历史之行命令
echo > ./.bash_history # 或者，清空用户目录下的这个文件即可
```

# 清楚历史执行命令

```c
vi /root/history #新建记录文件
history -c #清除记录
history -r /root/history.txt #导入记录
```

# example

```c
vi /root/history
history -c
history -r /root/history.txt
history
echo > /var/log/wtmp  
last
echo > /var/log/btmp
lastb
history -c
echo > ./.bash_history
history #查询导入结果
```
