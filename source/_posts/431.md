/**
 * 还原科学计数法
 */
public function scToNum($num)
{
    $num = floatval($num);
    $parts = explode('E', $num);
    if (count($parts) != 2) {
        return $num;
    }
    $exp = abs(end($parts)) + 3;
    $decimal = number_format($num, $exp);
    $decimal = rtrim($decimal, '0');

    $toNumber = rtrim($decimal, '.');

    return implode('', explode(',', $toNumber));
}
